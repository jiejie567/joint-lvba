cmake_minimum_required(VERSION 3.16)
project(Global-LVBA LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
include(CTest)
option(JOINT_LVBA_BUILD_PIPELINE "Build the ROS/PCL offline pipeline" ON)
find_package(Eigen3 REQUIRED)
find_package(Ceres 2.1 REQUIRED)

if(BUILD_TESTING)
  add_executable(joint_cost_test tests/test_joint_costs.cpp)
  target_include_directories(joint_cost_test PRIVATE include)
  target_link_libraries(joint_cost_test PRIVATE Ceres::ceres Eigen3::Eigen)
  add_test(NAME joint_costs COMMAND joint_cost_test)
endif()

if(JOINT_LVBA_BUILD_PIPELINE)
  find_package(catkin REQUIRED COMPONENTS roscpp std_msgs sensor_msgs pcl_conversions)
  find_package(OpenMP REQUIRED)
  find_package(PCL REQUIRED)
  find_package(OpenCV REQUIRED)
  find_package(SQLite3 REQUIRED)
  find_package(fmt REQUIRED)
  find_path(SOPHUS_INCLUDE_DIR NAMES sophus_lib/se3.hpp
    DOC "Directory containing the validated sophus_lib/ header directory")
  if(NOT SOPHUS_INCLUDE_DIR)
    message(FATAL_ERROR "Set SOPHUS_INCLUDE_DIR to the parent of sophus_lib/se3.hpp (see README.md)")
  endif()
  include_directories(include ${SOPHUS_INCLUDE_DIR} ${catkin_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
  add_definitions(${PCL_DEFINITIONS})
  add_library(lvba_system src/lvba_system.cpp)
  target_compile_definitions(lvba_system PUBLIC ROOT_DIR="${CMAKE_CURRENT_SOURCE_DIR}/")
  target_link_libraries(lvba_system PUBLIC Ceres::ceres Eigen3::Eigen ${OpenCV_LIBRARIES} ${PCL_LIBRARIES} ${catkin_LIBRARIES} SQLite::SQLite3 OpenMP::OpenMP_CXX fmt::fmt)
  add_library(lvba_io src/dataset_io.cpp)
  target_link_libraries(lvba_io PUBLIC lvba_system)
  add_executable(lidar_visual_ba src/main.cpp)
  target_link_libraries(lidar_visual_ba PRIVATE lvba_io lvba_system)
endif()

# Some Conda toolchains need their own libgcc; native toolchains leave this empty.
set(JOINT_LVBA_LIBGCC "" CACHE FILEPATH "Optional toolchain libgcc_s.so.1")
if(JOINT_LVBA_LIBGCC)
  if(NOT EXISTS "${JOINT_LVBA_LIBGCC}")
    message(FATAL_ERROR "JOINT_LVBA_LIBGCC does not exist")
  endif()
  foreach(executable joint_cost_test lidar_visual_ba)
    if(TARGET ${executable})
      target_link_libraries(${executable} PRIVATE "${JOINT_LVBA_LIBGCC}")
    endif()
  endforeach()
endif()
